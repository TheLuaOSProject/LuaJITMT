import gdb,collections
exits=collections.Counter()
errors=collections.Counter()
starts=collections.Counter()
class ExitCount(gdb.Breakpoint):
 def stop(self):
  try:
   pair=(int(gdb.parse_and_eval('parent')),int(gdb.parse_and_eval('exitno')))
   exits[pair]+=1
   gdb.execute('set $stage_J=J',to_string=True)
  except Exception as e: print('exit counter:',e)
  return False
ExitCount('lj_trace_exit',internal=True)
class ErrorCount(gdb.Breakpoint):
 def stop(self):
  try:
   e=str(gdb.parse_and_eval('e'))
   J=gdb.parse_and_eval('J')
   line=int(J['pt'].dereference()['firstline'])
   errors[(e,line)]+=1
  except Exception as e: print('error counter:',e)
  return False
ErrorCount('lj_trace_err',internal=True)
class StartCount(gdb.Breakpoint):
 def stop(self):
  try:
   J=gdb.parse_and_eval('J')
   line=int(J['pt'].dereference()['firstline'])
   phase=int(gdb.parse_and_eval('$stage_g->gc2.phase'))
   parent=int(J['parent'])
   starts[(phase,line,parent==0)]+=1
  except Exception as e: print('start counter:',e)
  return False
StartCount('trace_start',internal=True)
end
